## 📊 Data Models/DTOs

```dart
class UserDTO {
  String id;
  String name;
  String phoneNumber;
  String? profileImage;
  String? email;
  DateTime createdAt;
  DateTime updatedAt;
}

class VehicleDTO {
  String id;
  String type;           // "Auto Rickshaw", "Magic", etc.
  String description;
  String? image;
  double baseFare;
  double perKmRate;
  int capacity;
  String? estimatedTime;
}

class DriverDTO {
  String id;
  String name;
  String phoneNumber;
  String licenseNumber;
  double? rating;
  String? profileImage;
  VehicleDTO? assignedVehicle;
  bool isOnline;
  DateTime createdAt;
}

class BookingDTO {
  String id;
  String userId;
  String? driverId;
  String vehicleId;
  String pickupLocation;
  double pickupLat;
  double pickupLng;
  String dropoffLocation;
  double dropoffLat;
  double dropoffLng;
  double distance;
  double totalFare;
  BookingStatus status;
  String? otp;
  DateTime createdAt;
  DateTime? startedAt;
  DateTime? completedAt;
  double? rating;
  String? review;
}

enum BookingStatus {
  pending,
  assigned,
  started,
  completed,
  cancelled
}
```
